Introduction

Node.js lets you run JavaScript on the server. Combined with Express, it is one of the fastest ways to build REST APIs, real-time services, and full-stack web applications.

Chapter 2

Why Node.js?

Node.js uses Chrome's V8 engine to execute JavaScript outside the browser. Its event-driven, non-blocking I/O model makes it efficient for I/O-heavy workloads like APIs and real-time chat.

When to use Node.js

  • Real-time applications (WebSocket, chat, live dashboards).
  • Serverless functions and microservices.
  • Command-line tools and build scripts.

Chapter 5

Asynchronous JavaScript

Node.js is built around asynchronous operations. Understanding callbacks, promises, and async/await is essential for clean server code. 

const fs = require("fs").promises;
// Callbacks
fs.readFile("file.txt", "utf8", (err, data) => {
  if (err) return console.error(err);
  console.log(data);
});
// Promises
fs.readFile("file.txt", "utf8")
  .then((data) => console.log(data))
  .catch((err) => console.error(err));
// Async/await (preferred)
async function readFile() {
  try {
    const data = await fs.readFile("file.txt", "utf8");
    console.log(data);
  } catch (err) {
    console.error(err);
  }
}
readFile();

Node.js async patterns

  • Prefer async/await for readability.
  • Always handle errors with try/catch .
  • Avoid callback pyramids of doom.
  • Use Promise.all() to run independent async tasks in parallel.

Chapter 7

Routing

Routes define how the server responds to different URLs and HTTP methods. Organize related routes with Express routers. 

const express = require("express");
const app = express();
app.use(express.json());
// Basic CRUD routes for users
app.get("/users", (req, res) => {
  res.json([{ id: 1, name: "Alice" }]);
});
app.get("/users/:id", (req, res) => {
  res.json({ id: req.params.id, name: "Alice" });
});
app.post("/users", (req, res) => {
  const { name } = req.body;
  res.status(201).json({ id: 2, name });
});
app.put("/users/:id", (req, res) => {
  res.json({ id: req.params.id, name: req.body.name });
});
app.delete("/users/:id", (req, res) => {
  res.status(204).send();
});
// Query parameters
app.get("/search", (req, res) => {
  const { q } = req.query;
  res.json({ query: q });
});
app.listen(3000);

HTTP methods

  • GET: Read data.
  • POST: Create data.
  • PUT: Replace data.
  • PATCH: Partial update.
  • DELETE: Remove data.

Chapter 8

Middleware

Middleware functions run during request processing. They can modify requests, log activity, authenticate users, or handle errors. 

const express = require("express");
const app = express();
// Custom logger middleware
function logger(req, res, next) {
  console.log(`${req.method} ${req.url} - ${new Date().toISOString()}`);
  next();
}
app.use(logger);
// Error-handling middleware
function errorHandler(err, req, res, next) {
  console.error(err.stack);
  res.status(500).json({ error: err.message || "Internal Server Error" });
}
app.get("/", (req, res) => {
  res.json({ ok: true });
});
app.get("/error", (req, res, next) => {
  next(new Error("Something went wrong"));
});
app.use(errorHandler);
app.listen(3000);

Chapter 10

Full CRUD API

Combine Express routing with database queries to build a complete REST API for a resource. 

const express = require("express");
const db = require("./db");
const router = express.Router();
router.get("/", async (req, res, next) => {
  try {
    const { rows } = await db.query("SELECT * FROM products ORDER BY id");
    res.json(rows);
  } catch (err) {
    next(err);
  }
});
router.get("/:id", async (req, res, next) => {
  try {
    const { rows } = await db.query("SELECT * FROM products WHERE id = $1", [
      req.params.id,
    ]);
    if (rows.length === 0) return res.status(404).json({ error: "Not found" });
    res.json(rows[0]);
  } catch (err) {
    next(err);
  }
});
router.post("/", async (req, res, next) => {
  try {
    const { name, price, stock_quantity, category_id } = req.body;
    const { rows } = await db.query(
      "INSERT INTO products (name, price, stock_quantity, category_id) VALUES ($1, $2, $3, $4) RETURNING *",
      [name, price, stock_quantity, category_id],
    );
    res.status(201).json(rows[0]);
  } catch (err) {
    next(err);
  }
});
router.put("/:id", async (req, res, next) => {
  try {
    const { name, price, stock_quantity, category_id } = req.body;
    const { rows } = await db.query(
      "UPDATE products SET name = $1, price = $2, stock_quantity = $3, category_id = $4 WHERE id = $5 RETURNING *",
      [name, price, stock_quantity, category_id, req.params.id],
    );
    res.json(rows[0]);
  } catch (err) {
    next(err);
  }
});
router.delete("/:id", async (req, res, next) => {
  try {
    await db.query("DELETE FROM products WHERE id = $1", [req.params.id]);
    res.status(204).send();
  } catch (err) {
    next(err);
  }
});
module.exports = router;

const express = require("express");
const productsRouter = require("./routes/products");
const app = express();
app.use(express.json());
app.use("/products", productsRouter);
app.use((err, req, res, next) => {

Chapter 11

Validation & Error Handling

Never trust user input. Validate request data before touching the database and return helpful error responses. 

npm install express-validator
const { body, validationResult } = require("express-validator");
function validateProduct(req, res, next) {
  return [
    body("name").notEmpty().withMessage("Name is required"),
    body("price")
      .isFloat({ min: 0 })
      .withMessage("Price must be a positive number"),
    body("stock_quantity")
      .isInt({ min: 0 })
      .withMessage("Stock must be a non-negative integer"),
  ];
}
function handleValidationErrors(req, res, next) {
  const errors = validationResult(req);
  if (!errors.isEmpty()) {
    return res.status(400).json({ errors: errors.array() });
  }
  next();
}
router.post(
  "/",
  validateProduct(),
  handleValidationErrors,
  async (req, res, next) => {
    // create product...
  },
);

Validation best practices

  • Validate types, lengths, and ranges.
  • Sanitize strings to prevent injection and XSS.
  • Return 400 Bad Request for client errors.
  • Use a centralized error handler for unexpected errors.

Chapter 12

Authentication with JWT

JSON Web Tokens (JWT) are a common way to authenticate API users. After login, the client sends the token in the Authorization header. 

npm install jsonwebtoken bcryptjs
const jwt = require("jsonwebtoken");
const bcrypt = require("bcryptjs");
const JWT_SECRET = process.env.JWT_SECRET || "change-me";
function generateToken(user) {
  return jwt.sign({ id: user.id, email: user.email }, JWT_SECRET, {
    expiresIn: "1h",
  });
}
function authenticateToken(req, res, next) {
  const authHeader = req.headers.authorization;
  const token = authHeader && authHeader.split(" ")[1];
  if (!token) return res.status(401).json({ error: "Access denied" });
  jwt.verify(token, JWT_SECRET, (err, user) => {
    if (err) return res.status(403).json({ error: "Invalid token" });
    req.user = user;
    next();
  });
}
// Login route example
app.post("/login", async (req, res, next) => {
  const { email, password } = req.body;
  // fetch user hash from DB, compare with bcrypt
  const token = generateToken({ id: 1, email });
  res.json({ token });
});
// Protected route
app.get("/profile", authenticateToken, (req, res) => {
  res.json({ user: req.user });
});

Chapter 15

Capstone: Task API

Build a complete REST API for managing tasks. The API supports creating, reading, updating, deleting, and filtering tasks, backed by PostgreSQL.

Requirements

  • Schema: A tasks table with id, title, description, status, priority, due_date, and created_at.
  • CRUD: Full endpoints for tasks.
  • Filtering: Filter by status and priority via query parameters.
  • Validation: Title is required; status must be one of todo , in_progress , done .
  • Persistence: Store tasks in PostgreSQL.
  • Testing: At least one test for creating and listing tasks.

Schema 

CREATE TABLE tasks (
  id SERIAL PRIMARY KEY,
  title VARCHAR(255) NOT NULL,
  description TEXT,
  status VARCHAR(50) DEFAULT 'todo' CHECK (status IN ('todo', 'in_progress', 'done')),
  priority VARCHAR(50) DEFAULT 'medium' CHECK (priority IN ('low', 'medium', 'high')),
  due_date DATE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Routes 

// routes/tasks.js
const express = require("express");
const db = require("../db");
const router = express.Router();
router.get("/", async (req, res, next) => {
  try {
    const { status, priority } = req.query;
    let sql = "SELECT * FROM tasks WHERE 1=1";
    const params = [];
    if (status) {
      sql += " AND status = $1";
      params.push(status);
    }
    if (priority) {
      sql += ` AND priority = $${params.length + 1}`;
      params.push(priority);
    }
    sql += " ORDER BY created_at DESC";
    const { rows } = await db.query(sql, params);
    res.json(rows);
  } catch (err) {
    next(err);
  }
});
router.get("/:id", async (req, res, next) => {
  try {
    const { rows } = await db.query("SELECT * FROM tasks WHERE id = $1", [
      req.params.id,
    ]);
    if (rows.length === 0)
      return res.status(404).json({ error: "Task not found" });
    res.json(rows[0]);
  } catch (err) {
    next(err);
  }
});
router.post("/", async (req, res, next) => {
  try {
    const { title, description, status, priority, due_date } = req.body;
    if (!title) return res.status(400).json({ error: "Title is required" });
    const { rows } = await db.query(
      "INSERT INTO tasks (title, description, status, priority, due_date) VALUES ($1, $2, $3, $4, $5) RETURNING *",
      [title, description, status || "todo", priority || "medium", due_date],
    );
    res.status(201).json(rows[0]);
  } catch (err) {
    next(err);
  }
});
router.put("/:id", async (req, res, next) => {
  try {
    const { title, description, status, priority, due_date } = req.body;
    const { rows } = await db.query(
      "UPDATE tasks SET title = $1, description = $2, status = $3, priority = $4, due_date = $5 WHERE id = $6 RETURNING *",
      [title, description, status, priority, due_date, req.params.id],
    );
    res.json(rows[0]);
  } catch (err) {
    next(err);
  }
});
router.delete("/:id", async (req, res, next) => {
  try {
    await db.query("DELETE FROM tasks WHERE id = $1", [req.params.id]);
    res.status(204).send();
  } catch (err) {
    next(err);
  }
});
module.exports = router;
