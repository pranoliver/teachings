Chapter 2

Why TypeScript?

JavaScript is flexible but that flexibility can lead to runtime bugs. TypeScript catches many of those errors before your code runs.

JavaScript vs TypeScript example 

// JavaScript
function add(a, b) {
  return a + b;
}
add(2, "2"); // "22" — silent bug
// TypeScript
function add(a: number, b: number): number {
  return a + b;
}
add(2, "2"); // ❌ Error at compile time

Real-world benefits

  • Catch bugs before deployment.
  • Autocomplete and inline documentation in VS Code.
  • Safer refactoring across large codebases.
  • Self-documenting function contracts.
  • Widely used in React, Node.js, and full-stack frameworks.

Chapter 4

Basic Types

TypeScript provides types for the most common JavaScript values. Many types can be inferred automatically. 

// Primitives
const firstName: string = "Ada";
const age: number = 28;
const isActive: boolean = true;
// Type inference
const lastName = "Lovelace"; // inferred as string
// Any and unknown
let notSure: any = 4;
notSure = "maybe";
let safeValue: unknown = 4;
if (typeof safeValue === "number") {
  console.log(safeValue + 1); // safe narrowing
}
// Void and never
function logMessage(message: string): void {
  console.log(message);
}
function throwError(message: string): never {
  throw new Error(message);
}

Real-world tip

Avoid any whenever possible. Use unknown when you do not know the type yet, then narrow it with checks.

Chapter 5

Functions

Annotate function parameters and return values to make contracts explicit. Type inference also works for simple return expressions. 

// Named parameters and return type
function calculateTotal(price: number, quantity: number): number {
  return price * quantity;
}
// Optional parameter
function greetUser(name: string, greeting?: string): string {
  return `${greeting ?? "Hello"}, ${name}`;
}
// Default value
function createUser(name: string, role: string = "user") {
  return { name, role };
}
// Arrow function
const isPositive = (n: number): boolean => n > 0;
// Rest parameters
function sum(...numbers: number[]): number {
  return numbers.reduce((total, n) => total + n, 0);
}

Chapter 6

Objects & Interfaces

Interfaces describe the shape of objects. They are one of TypeScript's most powerful features for modeling domain data. 

interface Product {
  id: number;
  name: string;
  price: number;
  inStock: boolean;
}
function formatProduct(product: Product): string {
  return `${product.name} — $${product.price}`;
}
const laptop: Product = {
  id: 1,
  name: "MacBook Air",
  price: 1199,
  inStock: true,
};
console.log(formatProduct(laptop));
// Optional and readonly properties
interface User {
  readonly id: number;
  email: string;
  nickname?: string;
}
// Extending interfaces
interface Animal {
  name: string;
}
interface Dog extends Animal {
  breed: string;
}
const myDog: Dog = { name: "Rex", breed: "Labrador" };

When to use interfaces vs types

  • Use interface for object shapes and public APIs.
  • Use type for unions, aliases, and complex mapped types.
  • Interfaces can be extended and merged; type aliases cannot.

Chapter 7

Classes & OOP

TypeScript brings classic OOP features like access modifiers, abstract classes, and interfaces to JavaScript classes. 

class Account {
  private balance: number;
  constructor(
    public owner: string,
    initialBalance: number,
  ) {
    this.balance = initialBalance;
  }
  deposit(amount: number): void {
    if (amount <= 0) throw new Error("Amount must be positive");
    this.balance += amount;
  }
  getBalance(): number {
    return this.balance;
  }
}
const myAccount = new Account("Alice", 100);
myAccount.deposit(50);
console.log(myAccount.getBalance()); // 150
// myAccount.balance is private — access denied
abstract class Shape {
  abstract area(): number;
}
class Circle extends Shape {
  constructor(private radius: number) {
    super();
  }
  area(): number {
    return Math.PI * this.radius ** 2;
  }
}

Chapter 8

Arrays, Tuples & Enums

Typed arrays prevent mixed data mistakes. Tuples enforce fixed lengths, and enums group related constants. 

// Arrays
const scores: number[] = [85, 92, 78];
const names: Array = ["Ada", "Grace", "Linus"];
// Tuples (fixed length, fixed types)
const point: [number, number] = [10, 20];
const userTuple: [number, string, boolean] = [1, "Ada", true];
// Enums
enum OrderStatus {
  Pending = "pending",
  Shipped = "shipped",
  Delivered = "delivered",
}
function updateStatus(status: OrderStatus) {
  console.log(`Order is now ${status}`);
}
updateStatus(OrderStatus.Shipped);

Real-world note

Modern code often prefers string literal unions over enums because they compile to simpler JavaScript and work better with narrowing.

Chapter 9

Generics

Generics let you write reusable code that works with multiple types while preserving type safety. 

// Generic function
function identity(value: T): T {
  return value;
}
const num = identity(42);
const str = identity("hello"); // inferred
// Generic with constraints
function logLength(item: T): void {
  console.log(item.length);
}
logLength("hello");
logLength([1, 2, 3]);
// Generic interface
interface ApiResponse {
  data: T;
  success: boolean;
  message?: string;
}
const productResponse: ApiResponse = {
  data: laptop,
  success: true,

Chapter 10

Unions & Intersections

Union types let a value be one of several types. Intersections combine types into one. 

type Status = "loading" | "success" | "error";
function getMessage(status: Status): string {
  switch (status) {
    case "loading":
      return "Loading...";
    case "success":
      return "Done!";
    case "error":
      return "Something went wrong.";
  }
}
// Discriminated union
interface PayPalPayment {
  type: "paypal";
  email: string;
}
interface CardPayment {
  type: "card";
  cardNumber: string;
  expiry: string;
}
type PaymentMethod = PayPalPayment | CardPayment;
function processPayment(method: PaymentMethod) {
  if (method.type === "paypal") {
    console.log(method.email);
  } else {
    console.log(method.cardNumber);
  }
}
// Intersection
interface Named {
  name: string;
}
interface Dated {
  createdAt: Date;
}
type DocumentItem = Named & Dated;

Chapter 11

Utility Types

TypeScript includes built-in utility types that transform existing types. They are extremely useful for forms, APIs, and data shaping. 

interface User {
  id: number;
  email: string;
  name: string;
  isAdmin: boolean;
}
// Make all properties optional
type PartialUser = Partial;
// Make all properties required
type RequiredUser = Required;
// Pick only specific keys
type UserPreview = Pick;
// Exclude a key
type UserPublic = Omit;
// Readonly wrapper
type ReadonlyUser = Readonly;
// Record for mapped objects
const roles: Record = {
  admin: ["read", "write", "delete"],
  user: ["read"],
};

Common use cases

  • Partial for update forms and PATCH requests.
  • Pick for DTOs returned from an API.
  • Omit to hide internal fields in public responses.
  • Record for dictionaries and lookup maps.

// src/index.ts
import express, { Request, Response, NextFunction } from "express";
interface Task {
  id: number;
  title: string;
  completed: boolean;
}
const tasks: Task[] = [{ id: 1, title: "Learn TypeScript", completed: false }];
const app = express();
app.use(express.json());
app.get("/tasks", (req: Request, res: Response) => {
  res.json(tasks);
});
app.post("/tasks", (req: Request, res: Response, next: NextFunction) => {
  try {
    const { title } = req.body;
    if (!title) throw new Error("Title is required");
    const task: Task = { id: Date.now(), title, completed: false };
    tasks.push(task);
    res.status(201).json(task);
  } catch (err) {
    next(err);
  }
});
app.listen(3000, () => console.log("Server running on http://localhost:3000"));

Chapter 15

Capstone: E-Commerce Cart

Build a fully typed shopping cart module. This capstone combines interfaces, generics, utility types, and type narrowing into a realistic e-commerce example.

Requirements

  • Define Product and CartItem interfaces.
  • Implement add, remove, update quantity, and clear methods.
  • Calculate totals with discounts and tax.
  • Use generics for a reusable storage layer.
  • Write at least one test or script that exercises the cart.

Data models 

// src/types.ts
export interface Product {
  id: string;
  name: string;
  price: number;
  category: "electronics" | "clothing" | "food";
}
export interface CartItem {
  product: Product;
  quantity: number;
}
export type CartSummary = {
  items: CartItem[];
  subtotal: number;
  discount: number;
  tax: number;
  total: number;
};

Cart implementation 

// src/cart.ts
import { Product, CartItem, CartSummary } from "./types";
export class ShoppingCart {
  private items: CartItem[] = [];
  add(product: Product, quantity: number = 1): void {
    if (quantity <= 0) throw new Error("Quantity must be positive");
    const existing = this.items.find((i) => i.product.id === product.id);
    if (existing) {
      existing.quantity += quantity;
    } else {
      this.items.push({ product, quantity });
    }
  }
  remove(productId: string): void {
    this.items = this.items.filter((i) => i.product.id !== productId);
  }
  updateQuantity(productId: string, quantity: number): void {
    if (quantity <= 0) {
      this.remove(productId);
      return;
    }
    const item = this.items.find((i) => i.product.id === productId);
    if (item) item.quantity = quantity;
  }
  clear(): void {
    this.items = [];
  }
  getSummary(taxRate: number = 0.08, discountRate: number = 0): CartSummary {
    const subtotal = this.items.reduce(
      (sum, item) => sum + item.product.price * item.quantity,
      0,
    );
    const discount = subtotal * discountRate;
    const taxable = subtotal - discount;
    const tax = taxable * taxRate;
    const total = taxable + tax;
    return {
      items: [...this.items],
      subtotal: Number(subtotal.toFixed(2)),
      discount: Number(discount.toFixed(2)),
      tax: Number(tax.toFixed(2)),
      total: Number(total.toFixed(2)),
    };
  }
}

Generic storage layer 

// src/storage.ts
export interface Repository {
  findById(id: string): T | undefined;
  findAll(): T[];
  save(item: T): void;
  delete(id: string): void;
}
export class MemoryRepository implements Repository {
  private data: Map = new Map();
  findById(id: string): T | undefined {
    return this.data.get(id);
  }
  findAll(): T[] {
    return Array.from(this.data.values());
  }
  save(item: T): void {
    this.data.set(item.id, item);
  }
  delete(id: string): void {
    this.data.delete(id);
  }
}
